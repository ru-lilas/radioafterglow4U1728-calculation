\section{Distribution of Electrons}
We consider a distribution of post-shock electrons
with a number density \(n_{\ele}\).
We assume that the electron distribution consists of
the sum of thermal and non-thermal components.

\begin{equation}
  \ab(\pdv{n}{\gamma})
  =
  \ab(\pdv{n}{\gamma})_{\thermal}
  +
  \ab(\pdv{n}{\gamma})_{\powerlaw}
\end{equation}
\begin{equation}
  f_e
  \coloneq
  \frac{n_{\ele,\powerlaw}}{n_{\ele}}
  =
  \frac{
    n_{\ele,\powerlaw}
  }{
    n_{\ele,\thermal}+n_{\ele,\powerlaw}
  }
\end{equation}

Maxwell-J\"{u}ttner distribution
\(\Theta \coloneq k_{\symup{B}}T_{\ele}/(\melecc)\)
\begin{equation}
  \frac{1}{n_{\ele}}\ab(\pdv{n}{\gamma})_{\thermal}
  =
  (1-f_{\ele})
  f(\Theta)
  \sqrt{1-\gamma^{-2}}
  \frac{\gamma^2}{2\Theta^3}
  \exp\ab[-\frac{\gamma}{\Theta}]
\end{equation}
\begin{equation}
  f(\Theta)
  \coloneq
  \frac{2\Theta^2}{K_2(1/\Theta)}
\end{equation}
\begin{equation}
  \label{eq:thermal_energy}
  u_{\ele,\thermal} = a(\Theta)n_{\ele}\Theta \melecc
\end{equation}
\begin{equation}
  a(\Theta) 
  =
  \frac{1}{\Theta}
  \ab(\frac{K_3(1/\Theta)}{K_2(1/\Theta)}-1)
  -1
  \approx
  \frac{6+15\Theta}{4+5\Theta}
\end{equation}
We use the last approximate expression
\cite{gammieAdvectiondominatedAccretionFlows1998}
.

If a fraction $\epsth$ of the post-shock internal energy
is transferred to the thermal component, then we have
\begin{equation}
  \label{eq:thermal_partition}
  u_{\ele,\thermal} = \epsth u_{\shock}
\end{equation}
From Eqs.
\eqref{eq:thermal_energy}
and
\eqref{eq:thermal_partition}
,
\begin{equation}
  \label{eq:definition_of_theta}
  \Theta = \frac{5\Theta_0 - 6 + \sqrt{25\Theta_0^2 + 180\Theta_0 + 36}}{30}
  ,
\end{equation}
where
\begin{equation}
  \label{eq:definition_of_theta0}
  \Theta_0
  = 
  \frac{
    9
  }{
    32
  }
  \frac{
    \mu\mpro
  }{
    \mue\mele
  }
  \epsth\betash^2
  = \num{9.30}
  ~\ab(\frac{\epsth}{0.4})^2
  ~\ab(\frac{\betash}{0.3})^2
  ~\ab(\frac{\mue/\mu}{\num{2.0}})^{-1}
\end{equation}

Power-law distribution\((p>2)\)
\begin{empheq}[
    left={
      \displaystyle
      \frac{1}{n_{\ele}}\ab(\pdv{n}{\gamma})_{\powerlaw}
      =\empheqlbrace
    }
  ]{equation}
\begin{aligned}
  &f_{\ele}
    g_N(\Theta)
    \frac{p-2}{3\Theta}
    \ab(\frac{\gamma}{3\Theta})^{-p}
    &(\gamma \geq \gamma_{\min})\\
  &0
    &(\gamma < \gamma_{\min})
\end{aligned}
\end{empheq}
\begin{equation}
\end{equation}
\begin{equation}
  g_N(\Theta)
  \coloneq
  \frac{p-1}{p-2}
  \ab(\frac{\gamma_{\min}}{3\Theta})^{p-1}
\end{equation}
We assume
\begin{equation}
  \gamma_{\min}
  =
  \ab<\gamma>_{\thermal}
\end{equation}
The mean Lorentz factor of electrons following
a Maxwell-J\"{u}ttner distribution is given exactly by
\begin{equation}
  \ab<\gamma>_{\thermal}
  =
  1 + a(\Theta)\Theta
\end{equation}

\section*{}
Let \(\ab<E_{\thermal}>\) and \(\ab<E_{\mathrm{nth}}>\)
denote the mean kinetic energies per
thermal and non-thermal electron, respectively.
We define a fraction of kinetic energy density
of thermal electrons to non-thermal electrons:
\begin{equation}
  \delta
  \coloneq
  \frac{
    n_{\mathrm{nth}}\ab<E_{\mathrm{nth}}>
  }{
    n_{\thermal}\ab<E_{\thermal}>
  }
\end{equation}
\begin{equation}
  f_{\ele}
  \coloneq
  \frac{n_{\mathrm{nth}}}{n_{\mathrm{nth}}+n_{\thermal}}
\end{equation}
So
\begin{equation}
  \delta =
  \frac{f_{\ele}}{1-f_{\ele}}
  \ab(
    \frac{p-1}{p-2} + \frac{1}{(p-2)(\ab<\gamma>_{\thermal}-1)}
  )
\end{equation}
