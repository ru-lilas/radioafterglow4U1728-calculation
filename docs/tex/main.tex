\section{}
\(\epsB = \num{1.0e-2},\epsth = \num{0.40} \) are set,
and we assume ambient gas is composed of fully ionized hydrogen,
that is, \( \mu = \num{0.5}, \mue = \num{1.0}\).
\begin{equation}
  \muep \coloneq \mue/\mupro
\end{equation}
\begin{equation}
  \Mep
  \coloneq
  \frac{\mue\mele}{\mu\mpro}
  =
  \num{1.1e-3}
  ~\ab(\frac{\mue}{1.0})
  ~\ab(\frac{\mu}{0.5})^{-1}
\end{equation}
\begin{equation}
  R = \betash c t
\end{equation}
\begin{equation}
  \rho_{\wind} 
  \coloneq \frac{a^2\Acyc}{4\pi R^2} 
  = \frac{\Acyc}{4\pi}\ab(\frac{a}{\betash c t})^2
\end{equation}
\begin{equation}
  \phiA
  \coloneq 
  \sqrt{\frac{\Awind}{\Acyc}}
  =
  \qty{2.2e-1}{\GHz\min}
  ~\ab(\frac{\Awind}{\qty{1.0e+07}{\g\per\cm}})^{1/2}
\end{equation}
\begin{equation}
  \Acyc
  \coloneq 
  \ab(\frac{4\pi\melecc}{3ec})^2
  =
  \qty{5.672e-14}{\g\per\cm}
\end{equation}
\begin{equation}
  n_{\wind} 
  \coloneq \frac{\rho_{\wind}}{\mu \mpro} 
  = \frac{\Acyc}{4\pi\mu\mpro}
  \ab(\frac{a}{\betash c t})^2
\end{equation}
\begin{equation}
  n_{\ele} 
  \coloneq 4\mu_{\ele} n_{\wind}
  =
  \frac{\Acyc}{\pi\mele}
  \Mep
  \ab(\frac{a}{\betash c t})^2
\end{equation}
\begin{equation}
  \frac{B^2}{8\pi}
  = \epsB \frac{9}{8}\rho_{\wind}(\betash c)^2
\end{equation}
\begin{equation}
  B
  = 
  \frac{3}{2}
  \ab(\epsB \Acyc)^{1/2}
  \frac{a}{t}
\end{equation}
\begin{equation}
  u_{\ele} = a(\Theta)n_{\ele}\Theta \melecc
\end{equation}
\begin{equation}
  u = \frac{9}{8}n\mu \mpro (\betash c)^2
\end{equation}
\begin{equation}
  u_{\ele} = \epsth u
\end{equation}
\begin{equation}
  \label{eq:definition_of_theta}
  \Theta = \frac{5\Theta_0 - 6 + \sqrt{25\Theta_0^2 + 180\Theta_0 + 36}}{30}
  \underset{\Theta_0\gg 1}{\approx}
  \frac{1}{3}\Theta_0
\end{equation}
\begin{equation}
  \label{eq:definition_of_theta0}
  \Theta_0
  = 
  \frac{
    9
  }{
    32
  }
  \frac{
    \epsth\betash^2
  }{
    \Mep
  }
  = \num{9.30}
  ~\ab(\frac{\epsth}{0.4})^2
  ~\ab(\frac{\betash}{0.3})^2
  ~\ab(\frac{\Mep}{\num{1.1e-03}})^{-1}
\end{equation}
\begin{equation}
  \label{eq:theta_in_hotlimit}
  \Theta
  \underset{\Theta_0\gg 1}{\approx}
  \frac{
    3
  }{
    32
  }
  \frac{
    \epsth\betash^2
  }{
    \Mep
  }
\end{equation}
\begin{equation}
  \label{eq:definition_of_phi_theta}
  \phith
  =
  \frac{3}{2}\Theta^2 \phiA\sqrt{\epsB}
\end{equation}
\begin{equation}
  \label{eq:definition_of_j_nu}
  j_{\nu}
  =
  \frac{
    \sqrt{3} e^3 n_{\ele} B
  }{
    8\pi \melecc
  }
  f(\Theta)
  =
  \frac{\pi}{
    8\sqrt{3}
  }
  \frac{
    \epsth\sqrt{\epsB}
  }{
    \Theta
  }
  \melecc
  \ab(\frac{a}{ct})^3
  f(\Theta)
\end{equation}
\begin{equation}
  \label{eq:definition_of_alpha_nu}
  \alpha_{\nu}
  =
  \frac{
    \pi e n_{\ele}
  }{
    3^{3/2}\Theta^5 B
  }
  f(\Theta)
  =
  \frac{
    \pi
  }{
    4\cdot 3^{5/2}
  }
  \frac{
    \epsth
  }{
    \Theta^6\sqrt{\epsB}
  }
  \ab(\frac{a}{ct})
  f(\Theta)
\end{equation}
\begin{equation}
  \label{eq:definition_of_tau_theta}
  \tauth
  \coloneq
  \alpha_{\nu}R
  =
  \frac{\phiA}{27}
  \sqrt{
    \frac{
      2\Mep
    }{
      \epsB\Theta^{11}
    }
  }
  f(\Theta)
\end{equation}
\begin{equation}
  \label{eq:definition_of_l_theta}
  \lnuth
  \coloneq
  4\pi^2 R^2
  \frac{
    j_{\Theta}
  }{
    \alpha_{\Theta}
  }
  =
  192\pi^3
  \Theta^6
  \epsth^{-1}
  \epsB
  \Mep
  \phiA^2
  \melecc
\end{equation}

Here, we treated the persistent emission as
the observed points
before the onset of Type I X-ray burst,
and the value of the persistent flux was
taken by the mean of these observed points.

The peak flux was estimated
by taking the mean of the two heighest values,
and the peak time was defined as the midpoint
between the corresponding points.

\begin{equation}
  \phi_{\peak} = \phi_{\Theta}\xm^{\peak}
\end{equation}
\begin{equation}
  f_{\nu,\peak} = 
  \delta^3 \frac{L_{\Theta}}{4\pi D^2}
  \Lambda_{\tau_{\Theta},\peak}
\end{equation}
